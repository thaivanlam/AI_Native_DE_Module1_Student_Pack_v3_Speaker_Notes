SELECT TO_CHAR(DATE_TRUNC('month', order_date AT TIME ZONE 'Asia/Ho_Chi_Minh'), 'YYYY-MM') AS order_month,
       COUNT(*)         AS order_count,
       SUM(order_total) AS total_revenue
FROM core.orders
WHERE order_status <> 'cancelled'
GROUP BY 1
ORDER BY 1;
